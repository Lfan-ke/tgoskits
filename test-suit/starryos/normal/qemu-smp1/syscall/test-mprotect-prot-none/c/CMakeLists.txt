cmake_minimum_required(VERSION 3.20)
project(test-mprotect-prot-none C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-mprotect-prot-none src/main.c)
target_include_directories(test-mprotect-prot-none PRIVATE src)
target_compile_options(test-mprotect-prot-none PRIVATE -Wall -Wextra -Werror)
install(TARGETS test-mprotect-prot-none RUNTIME DESTINATION usr/bin)
