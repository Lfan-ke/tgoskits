cmake_minimum_required(VERSION 3.20)
project(test-times C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
add_executable(test-times src/main.c)
target_include_directories(test-times PRIVATE src)
target_compile_options(test-times PRIVATE -Wall -Wextra)
target_link_libraries(test-times PRIVATE Threads::Threads)
install(TARGETS test-times RUNTIME DESTINATION usr/bin/starry-test-suit)
