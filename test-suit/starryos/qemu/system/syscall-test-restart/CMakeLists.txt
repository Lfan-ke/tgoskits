cmake_minimum_required(VERSION 3.20)
project(test-restart C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test-restart src/main.c)
target_include_directories(test-restart PRIVATE src)
target_compile_options(test-restart PRIVATE -Wall -Wextra)
install(TARGETS test-restart RUNTIME DESTINATION usr/bin/starry-test-suit)
